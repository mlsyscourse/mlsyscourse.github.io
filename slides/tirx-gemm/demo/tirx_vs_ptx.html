// SmemDescriptor encoding (per operand)
uint64_t desc = base_addr;
desc |= ((uint64_t)(ldo >> 4)) << 16;
desc |= ((uint64_t)(sdo >> 4)) << 32;
desc |= ((uint64_t)swizzle_mode) << 62;

// InstrDescriptor encoding
uint32_t idesc = 0;
idesc |= (1u << 4);   // dtype=FP32
idesc |= (1u << 7);   // atype=FP16
idesc |= (1u << 10);  // btype=FP16
idesc |= (N >> 3) << 17;  // MMA_N
idesc |= (M >> 4) << 24;  // MMA_M

// mbarrier setup
mbarrier.init [mbar], 1;
fence.mbarrier_init.release.cluster;

// Issue MMA (single thread)
tcgen05.mma.cta_group::1.kind::f16
    [tmem_addr], a_desc, b_desc, idesc, accum;

tcgen05.commit.cta_group::1
    .mbarrier::arrive::one [mbar];

// Wait for completion
mbarrier.try_wait.parity [mbar], phase;
