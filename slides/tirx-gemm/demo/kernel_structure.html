@Tx.prim_func(tirx=True)
def hgemm(A, B, D):
  with Tx.kernel():
    bx, by = Tx.cta_id([...], parent="kernel")
    wg_id = Tx.warpgroup_id([1], parent="cta")
    warp_id = Tx.warp_id([4], parent="warpgroup")
    lane_id = Tx.thread_id([32], parent="warp")
    pool = Tx.PoolAllocator()
    Asmem = pool.alloc((128,64), "fp16", layout=A_layout)
    Bsmem = pool.alloc((128,64), "fp16", layout=B_layout)
    pool.commit()
    Tx.ptx.tcgen05.alloc(tmem_addr, n_cols=512)
    with Tx.cta():              # all 128 threads
      Tx.copy(Asmem, A[m:m+128, 0:64])
      Tx.copy(Bsmem, B[n:n+128, 0:64])
    Tx.cuda.cta_sync()
    with Tx.thread()[elect_sync()]:  # 1 thread
      Tx.gemm_async(tmem, Asmem, Bsmem,
          accum=False, dispatch="tcgen05")
      Tx.ptx.tcgen05.commit(mma_bar, cta_group=1)
    Tx.ptx.mbarrier.try_wait(mma_bar, phase)
    with Tx.warpgroup():       # 128 threads
      Tx.copy(Dreg_wg, tmem[:,:BLK_N])
    with Tx.thread():           # per thread
      Tx.cast(Dreg_f16, Dreg)
      Tx.copy(D[m_thr, n:n+BLK_N], Dreg_f16)
